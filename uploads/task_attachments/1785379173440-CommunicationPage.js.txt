const express = require("express");
const router = express.Router();
const authMiddleware = require("../middleware/authMiddleware");
const requireRole = require("../middleware/roleMiddleware");
const {
  getConversations,
  getConversationWithUser,
  sendMessage,
  markMessageRead,
  markMessagesReadFromUser,
} = require("../controllers/messageController");

router.use(authMiddleware);

// Allow any authenticated user to access messaging endpoints
router.get("/", getConversations);
router.get("/user/:userId", getConversationWithUser);
router.post("/", sendMessage);
router.patch("/user/:userId/read", markMessagesReadFromUser);
router.patch("/:id/read", markMessageRead);

module.exports = router;
